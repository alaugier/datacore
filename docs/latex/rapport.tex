% Extraits de code : style sobre, numéros de ligne, pas de coloration
% agressive -- le code cite sa source réelle dans le dépôt, il n'a pas
% vocation à remplacer la lecture du fichier.
% showstringspaces=false : sans ça, listings affiche un symbole visible
% pour chaque espace a l'interieur d'une chaine Python (ex. un message
% d'erreur) -- rendu a tort comme des "caracteres bizarres" en relecture.

  basicstyle=\ttfamily\small,
  numbers=left,
  numberstyle=\tiny\color{gray},
  breaklines=true,
  showstringspaces=false,
  frame=single,
  backgroundcolor=\color{gray!5},
  keywordstyle=\color{blue!60!black},
  commentstyle=\color{gray!60!black}\itshape,
  stringstyle=\color{green!40!black},
}

% Chemins de fichiers : \filepath (alias de \path, cassable aux "/")
% plutot que \texttt (qui ne casse jamais une ligne) -- sans ca, un
% chemin long deborde silencieusement dans la marge de droite.
% \DeclareRobustCommand (pas \newcommand) : \path casse sinon dans un
% argument mobile (ex. \caption{}) -- erreur fatale trouvée a la
% compilation apres l'avoir utilise dans une legende de figure.
\DeclareRobustCommand{\filepath}[1]{\path{#1}}
